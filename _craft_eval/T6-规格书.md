if pdf_path:
    from services.redact_audit import detect_faux_redactions
    ra = detect_faux_redactions(pdf_path)
    if not ra["ok"]:
        add("W7b.伪脱敏", "PDF可解析", False, f"PDF 无法解析: {ra['error']}", "high")
    else:
        add("W7b.伪脱敏", "PDF伪脱敏检测", ra["count"] == 0,
            (f"检出 {ra['count']} 处伪脱敏: {[f['covered_text'][:30] for f in ra['findings'][:3]]}" if ra["count"] else "未检出"),
            "high" if ra["count"] else "low")
if xlsx_path:
    from utils.craft_bridge import GridCraft
    gc = GridCraft().check(xlsx_path)
    if gc["verdict"] == "all_pass":
        add("W7b.计算校验", "gridcraft check", True, f"all_pass | {gc['detail']}", "low")
    elif gc["verdict"] == "mismatch":
        add("W7b.计算校验", "gridcraft check", False, f"公式重算不一致 | {gc['detail']}", "high")
    else:  # open_error / usage_error / NOEXE/超时等
